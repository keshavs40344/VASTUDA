  // ─── SUITE HUB DROPDOWN TOGGLE LOGIC ───
  (function() {
    const hubBtn = document.getElementById('btn-open-hub-menu');
    const hubMenu = document.getElementById('suite-dropdown-menu');
    if (hubBtn && hubMenu) {
      hubBtn.addEventListener('click', function(e) {
        e.stopPropagation();
        hubMenu.classList.toggle('show');
      });
      document.addEventListener('click', function(e) {
        if (!hubMenu.contains(e.target) && e.target !== hubBtn) {
          hubMenu.classList.remove('show');
        }
      });
      // Close dropdown when any item inside is clicked
      hubMenu.querySelectorAll('.suite-menu-item').forEach(item => {
        item.addEventListener('click', function() {
          hubMenu.classList.remove('show');
        });
      });
    }
  })();
