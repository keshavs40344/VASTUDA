      Knowledge Base
      
Frequently Asked Questions

      
Industry guidelines, photo conventions, and ATS parsing mechanics.

    


    

      
      

        
          Should I include a photograph on my resume or CV?
          
        
        

          
Resume photo expectations are not universal and depend strictly on your target country and industry:

          
    
            
  • United States, United Kingdom, Canada, Australia: Photographs should strictly not be included. Corporate ATS systems and hiring teams avoid photos to comply with anti-discrimination employment laws.
    • 
            
  • Continental Europe (e.g. Germany, France, Austria): Traditional corporate sectors have historically expected a professional headshot (e.g. German Bewerbungsfoto), although modern international tech startups frequently prefer photo-free resumes.
    • 
            
  • Middle East & Select Asian Regions: Photos are common in hospitality, client-facing, and regional corporate applications.
    • 
          

          
VASTUDA Solution: Toggle "Include Photo" ON or OFF with 1 click to match the precise convention of your target role.

        

      


      
      

        
          How does VASTUDA's ATS Compatibility Analysis work?
          
        
        

          
There is no single, universal "ATS algorithm"—different employers use distinct systems (Workday, Greenhouse, Lever, Taleo, iCIMS), each with its own parser. VASTUDA's ATS Compatibility Analysis scores your resume against the universal rules shared by all modern parsers: clean UTF-8 character extraction, canonical header labeling, absence of unparsable graphics/tables, and natural keyword density in measurable accomplishment statements.

        

      


      
      

        
          Is my personal data or resume uploaded to any external server?
          
        
        

          
No. We operate under a strict zero-cloud-egress client architecture. All personal details, document OCR, bullet optimizations, and PDF generation occur in your browser's local RAM and LocalStorage. You can also export a permanent local backup at any time using the Backup JSON button.

function downloadPortfolioHTML() {
  const p = STATE.profile;
  const frame = document.getElementById('portfolioPreviewFrame');
  if (!frame) return;

  const html = frame.srcdoc;
  const blob = new Blob([html], { type: 'text/html' });
  const a = document.createElement('a');
  a.href = URL.createObjectURL(blob);
  a.download = `portfolio_${(p.name || 'candidate').toLowerCase().replace(/\s+/g, '_')}.html`;
  a.click();
  showToast('Downloaded portfolio index.html! Ready to deploy.');
}

function selectCareerDocType(type) {
  STATE.activeDocType = type;
  ['cover', 'cold', 'thankyou', 'resignation'].forEach(t => {
    const btn = document.getElementById(`doc-btn-${t}`);
    if (btn) {
      if (t === type) {
        btn.className = 'px-3 py-1.5 rounded-lg bg-pink-500/20 text-pink-400 border border-pink-500/40 font-bold whitespace-nowrap text-xs';
      } else {
        btn.className = 'px-3 py-1.5 rounded-lg bg-slate-800 text-slate-400 hover:text-white whitespace-nowrap text-xs';
      }
    }
  });
  renderSelectedCareerDoc();
}

// Job Application Tracker CRM
function renderApplicationsList() {
  const container = document.getElementById('trackerTableBody');
  if (!container) return;

  const apps = STATE.applications || [];
  if (apps.length === 0) {
    container.innerHTML = `No applications tracked yet. Click "+ Add Application" above.`;
    return;
  }

  container.innerHTML = apps.map((app, idx) => {
    let statusClass = 'bg-slate-700 text-slate-300';
    if (app.status === 'Offer') statusClass = 'bg-emerald-500/20 text-emerald-400 border border-emerald-500/30';
    if (app.status === 'Interviewing') statusClass = 'bg-amber-500/20 text-amber-400 border border-amber-500/30';
    if (app.status === 'Applied') statusClass = 'bg-blue-500/20 text-blue-400 border border-blue-500/30';

    return `
      
        ${escapeHtml(app.company)}
        ${escapeHtml(app.role)}
        
          ${escapeHtml(app.status)}
        
        ${escapeHtml(app.date)}
        ${escapeHtml(app.salary || '-')}
        
          
        
      
    `;
  }).join('');
}

function addNewApplicationModal() {
  const company = prompt('Enter Target Company Name (e.g. Netflix, Microsoft):');
  if (!company) return;
  const role = prompt('Enter Target Job Title:', STATE.profile.title || 'Software Engineer');
  const salary = prompt('Expected Compensation (optional, e.g. $220k):', '$200k+');

  STATE.applications.unshift({
    id: Date.now(),
    company,
    role: role || 'Software Engineer',
    status: 'Applied',
    date: new Date().toISOString().split('T')[0],
    salary: salary || '-',
    notes: 'Submitted via company career portal.'
  });

  saveStateToLocal();
  renderApplicationsList();
  showToast(`Added ${company} to your Application Pipeline!`);
}

function deleteApplication(index) {
  STATE.applications.splice(index, 1);
  saveStateToLocal();
  renderApplicationsList();
  showToast('Application record removed.');
}

function formatBulletText(text) {
  if (!text) return '';
  const safe = escapeHtml(text);
  return safe.replace(/(\b\d+(\.\d+)?%|\$\d+([,\.]\d+)?[kmb]?|\b\d+x\b|\b\d{2,}\b)/gi, '$1');
}

function updatePageBreakIndicator() {
  const sheet = document.getElementById('resumeSheet');
  const indicator = document.getElementById('pageCountIndicator');
  if (!sheet || !indicator) return;

  const height = sheet.scrollHeight;
  const pages = Math.ceil(height / 1123);

  indicator.innerHTML = `
    
      
      ${pages === 1 ? '1-Page Clean Length' : `${pages} Pages Detected`}
    
  `;
}

  const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(STATE, null, 2));
  const a = document.createElement('a');
  a.href = dataStr;
  a.download = `vastuda_master_${(STATE.profile.name || 'candidate').toLowerCase().replace(/\s+/g, '_')}.json`;
  a.click();
  showToast('Exported Master Profile Backup JSON!');
}

function importMasterJSON(event) {
  const file = event.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = (e) => {
    try {
      const parsed = JSON.parse(e.target.result);
      if (parsed && parsed.profile) {
        STATE = parsed;
        hydrateFormFromState();
        renderExperienceFormBlocks();
        renderEducationFormBlocks();
        renderProjectsFormBlocks();
        renderResumeSheet();
        computeATSScoreQuick();
        showToast('Successfully restored master profile!');
      }
    } catch (err) {
      alert('Error parsing JSON backup file: ' + err.message);
    }
  };
  reader.readAsText(file);
}

function showToast(msg) {
  let toast = document.getElementById('vastuda-toast');
  if (!toast) {
    toast = document.createElement('div');
    toast.id = 'vastuda-toast';
    toast.className = 'fixed bottom-6 right-6 z-50 bg-slate-900/95 text-white text-xs font-semibold px-4 py-2.5 rounded-xl border border-amber-400/40 shadow-2xl flex items-center gap-2 transition-all duration-300 transform translate-y-10 opacity-0';
    document.body.appendChild(toast);
  }
  toast.innerHTML = ` ${escapeHtml(msg)}`;
  toast.classList.remove('translate-y-10', 'opacity-0');
  setTimeout(() => toast.classList.add('translate-y-10', 'opacity-0'), 3000);
}

function escapeHtml(str) {
  if (!str) return '';
  return String(str)
    .replace(/&/g, '&')
    .replace(//g, '>')
    .replace(/"/g, '"')
    .replace(/'/g, ''');
}
